~~~powershell
param(
    [string]$ApiUrl = 'http://localhost:8080',
    [string]$MailpitUrl = 'http://localhost:8025'
)
$ErrorActionPreference = 'Stop'
# Run only against local/dev services. No passwords or tokens are printed.
$taskSuffix = [Guid]::NewGuid().ToString('N').Substring(0, 12)
$taskEmail = "smoke-$taskSuffix@example.com"
$taskPhone = '09' + (Get-Random -Minimum 10000000 -Maximum 99999999)
$taskPassword = [Guid]::NewGuid().ToString('N') + 'Aa1!'
function Post-Json([string]$Path, $Body) {
    Invoke-RestMethod -Method Post -Uri "$ApiUrl$Path" -ContentType 'application/json' -Body ($Body | ConvertTo-Json -Compress)
}
$null = Post-Json '/auth/signup' @{
    fullName = 'Auth Smoke Test'; email = $taskEmail; phone = $taskPhone
    password = $taskPassword; confirmPassword = $taskPassword; termsAccepted = $true; role = 'ADMIN'
}
$taskMessage = $null
for ($attempt = 0; $attempt -lt 20; $attempt++) {
    $messages = Invoke-RestMethod -Uri "$MailpitUrl/api/v1/messages"
    $taskMessage = $messages.messages | Where-Object {
        @($_.To | ForEach-Object { $_.Address }) -contains $taskEmail
    } | Select-Object -First 1
    if ($taskMessage) { break }
    Start-Sleep -Milliseconds 500
}
if (-not $taskMessage) { throw 'Verification email not found in Mailpit' }
$detail = Invoke-RestMethod -Uri "$MailpitUrl/api/v1/message/$($taskMessage.ID)"
$tokenMatch = [regex]::Match($detail.Text, 'token=([A-Za-z0-9_-]{43})')
if (-not $tokenMatch.Success) { throw 'Verification link not found' }
$null = Invoke-RestMethod -Uri "$ApiUrl/auth/verify-email?token=$($tokenMatch.Groups[1].Value)"
$pair = Post-Json '/auth/login' @{ email = $taskEmail; password = $taskPassword }
if ($pair.expiresIn -ne 900 -or $pair.refreshExpiresIn -ne 604800) { throw 'Unexpected token TTL' }
$rotated = Post-Json '/auth/refresh' @{ refreshToken = $pair.refreshToken }
if ($rotated.refreshToken -eq $pair.refreshToken) { throw 'Rotation failed' }
$null = Post-Json '/auth/logout' @{ refreshToken = $rotated.refreshToken }
try {
    $null = Post-Json '/auth/refresh' @{ refreshToken = $rotated.refreshToken }
    throw 'Revoked refresh token was accepted'
} catch {
    if (-not $_.Exception.Response -or [int]$_.Exception.Response.StatusCode -ne 401) { throw }
}
Write-Host 'PASS: signup -> email -> verify -> login -> rotate -> logout -> revoked token rejected.'
Write-Host "Development account created: $taskEmail"
~~~

~~~java
package com.kitolam.rentalroommanagement.common.base;

import lombok.Getter;
import lombok.Setter;
import org.springframework.data.annotation.CreatedDate;
import org.springframework.data.annotation.Id;
import org.springframework.data.annotation.LastModifiedDate;

import java.time.Instant;

/**
 * Base class for every MongoDB document.
 * <ul>
 *   <li>createdAt / updatedAt are filled by Spring Data auditing (see MongoConfig).</li>
 *   <li>Soft delete: isDeleted starts as false, {@link #markDeleted()} flips it and stamps deletedAt.
 *       Every read query must filter {@code isDeleted = false}.</li>
 * </ul>
 */
@Getter
@Setter
public abstract class BaseDocument {

    @Id
    private String id;

    @CreatedDate
    private Instant createdAt;

    @LastModifiedDate
    private Instant updatedAt;

    private Instant deletedAt = null;

    private Boolean isDeleted = false;

    public void markDeleted() {
        this.isDeleted = true;
        this.deletedAt = Instant.now();
    }
}
~~~

~~~java
package com.kitolam.rentalroommanagement.common.base;

import org.springframework.data.domain.Page;
import org.springframework.data.domain.Pageable;
import org.springframework.data.mongodb.core.query.Criteria;
import org.springframework.data.mongodb.repository.MongoRepository;
import org.springframework.data.repository.NoRepositoryBean;

import java.util.Optional;

/**
 * Shared repository for documents extending {@link BaseDocument}.
 * findWithPagination is implemented in {@link BaseRepositoryImpl}; every method here ignores soft-deleted documents.
 */
@NoRepositoryBean
public interface BaseRepository<T extends BaseDocument> extends MongoRepository<T, String> {

    Optional<T> findByIdAndIsDeletedFalse(String id);

    boolean existsByIdAndIsDeletedFalse(String id);

    /** Page of all non-deleted documents. */
    Page<T> findWithPagination(Pageable pageable);

    /**
     * Page of non-deleted documents matching {@code criteria} (null = no extra condition).
     * Reference fields are stored as ObjectId, so compare them with MongoQueryHelper.toObjectId(...).
     */
    Page<T> findWithPagination(Criteria criteria, Pageable pageable);
}
~~~

~~~java
package com.kitolam.rentalroommanagement.common.base;

import com.kitolam.rentalroommanagement.common.util.MongoQueryHelper;
import org.springframework.data.domain.Page;
import org.springframework.data.domain.Pageable;
import org.springframework.data.mongodb.core.MongoOperations;
import org.springframework.data.mongodb.core.query.Criteria;
import org.springframework.data.mongodb.core.query.Query;
import org.springframework.data.mongodb.repository.query.MongoEntityInformation;
import org.springframework.data.mongodb.repository.support.SimpleMongoRepository;
import org.springframework.data.support.PageableExecutionUtils;

import java.util.List;

/**
 * Base class behind every module repository (registered via repositoryBaseClass in MongoConfig).
 * Spring Data routes the custom methods declared in {@link BaseRepository} here by signature.
 */
public class BaseRepositoryImpl<T extends BaseDocument> extends SimpleMongoRepository<T, String> {

    private final MongoOperations mongoOperations;
    private final MongoEntityInformation<T, String> entityInformation;

    public BaseRepositoryImpl(MongoEntityInformation<T, String> entityInformation, MongoOperations mongoOperations) {
        super(entityInformation, mongoOperations);
        this.entityInformation = entityInformation;
        this.mongoOperations = mongoOperations;
    }

    public Page<T> findWithPagination(Pageable pageable) {
        return findWithPagination(null, pageable);
    }

    public Page<T> findWithPagination(Criteria criteria, Pageable pageable) {
        Criteria where = criteria == null
                ? MongoQueryHelper.notDeleted()
                : new Criteria().andOperator(MongoQueryHelper.notDeleted(), criteria);
        Query query = new Query(where).with(pageable);

        Class<T> type = entityInformation.getJavaType();
        String collection = entityInformation.getCollectionName();
        List<T> content = mongoOperations.find(query, type, collection);
        // Count runs only when needed (e.g. not on a last, partially filled page)
        return PageableExecutionUtils.getPage(content, pageable,
                () -> mongoOperations.count(Query.of(query).limit(-1).skip(-1), type, collection));
    }
}
~~~

~~~java
package com.kitolam.rentalroommanagement.common.exception.exceptions;

import lombok.Getter;
import org.springframework.http.HttpStatus;

/**
 * Base class for all business exceptions. Each subclass binds to one HTTP status,
 * so GlobalExceptionHandler only needs a single handler for the whole family.
 */
@Getter
public abstract class AppException extends RuntimeException {

    private final HttpStatus status;
    private final String errorCode;

    protected AppException(HttpStatus status, String message) {
        this(status, message, (String) null);
    }

    protected AppException(HttpStatus status, String message, String errorCode) {
        super(message);
        this.status = status;
        this.errorCode = errorCode;
    }

    protected AppException(HttpStatus status, String message, Throwable cause) {
        super(message, cause);
        this.status = status;
        this.errorCode = null;
    }
}
~~~

~~~java
package com.kitolam.rentalroommanagement.common.util;

import com.kitolam.rentalroommanagement.common.exception.exceptions.BadRequestException;
import org.bson.types.ObjectId;
import org.springframework.data.domain.Page;
import org.springframework.data.domain.Pageable;
import org.springframework.data.mongodb.core.MongoTemplate;
import org.springframework.data.mongodb.core.query.Criteria;
import org.springframework.data.mongodb.core.query.Query;
import org.springframework.data.support.PageableExecutionUtils;
import org.springframework.util.StringUtils;

import java.util.ArrayList;
import java.util.Arrays;
import java.util.List;
import java.util.regex.Pattern;

/**
 * Small helpers to build soft-delete-aware dynamic queries.
 * Reference fields are stored as ObjectId, so they must be compared with {@link #toObjectId(String)}.
 */
public final class MongoQueryHelper {

    public static final String IS_DELETED = "isDeleted";

    private MongoQueryHelper() {
    }

    public static Criteria notDeleted() {
        return Criteria.where(IS_DELETED).is(false);
    }

    public static List<Criteria> baseCriteria() {
        List<Criteria> criteria = new ArrayList<>();
        criteria.add(notDeleted());
        return criteria;
    }

    public static void addEquals(List<Criteria> criteria, String field, Object value) {
        if (value == null) {
            return;
        }
        criteria.add(Criteria.where(field).is(value instanceof Enum<?> e ? e.name() : value));
    }

    public static void addObjectIdEquals(List<Criteria> criteria, String field, String id) {
        if (StringUtils.hasText(id)) {
            criteria.add(Criteria.where(field).is(toObjectId(id)));
        }
    }

    public static void addKeyword(List<Criteria> criteria, String keyword, String... fields) {
        if (!StringUtils.hasText(keyword)) {
            return;
        }
        String regex = Pattern.quote(keyword.trim());
        Criteria[] matchers = Arrays.stream(fields)
                .map(field -> Criteria.where(field).regex(regex, "i"))
                .toArray(Criteria[]::new);
        criteria.add(new Criteria().orOperator(matchers));
    }

    public static ObjectId toObjectId(String id) {
        if (id == null || !ObjectId.isValid(id)) {
            throw new BadRequestException("Id không hợp lệ: " + id);
        }
        return new ObjectId(id);
    }

    public static <T> Page<T> page(MongoTemplate mongoTemplate, List<Criteria> criteria, Pageable pageable, Class<T> type) {
        Query query = new Query(new Criteria().andOperator(criteria)).with(pageable);
        List<T> content = mongoTemplate.find(query, type);
        return PageableExecutionUtils.getPage(content, pageable,
                () -> mongoTemplate.count(Query.of(query).limit(-1).skip(-1), type));
    }

    /**
     * True if another non-deleted document (other than {@code excludeId}) matches all conditions.
     * Used for business uniqueness, since soft-deleted rows must not block re-use of a value.
     */
    public static boolean existsOther(MongoTemplate mongoTemplate, Class<?> type, String excludeId, Criteria... conditions) {
        List<Criteria> criteria = baseCriteria();
        criteria.addAll(Arrays.asList(conditions));
        if (excludeId != null) {
            criteria.add(Criteria.where("id").ne(toObjectId(excludeId)));
        }
        return mongoTemplate.exists(new Query(new Criteria().andOperator(criteria)), type);
    }
}
~~~

~~~java
package com.kitolam.rentalroommanagement.modules.auth.repository;

import com.kitolam.rentalroommanagement.modules.auth.document.*;
import lombok.RequiredArgsConstructor;
import org.bson.types.ObjectId;
import org.springframework.data.mongodb.core.MongoTemplate;
import org.springframework.data.mongodb.core.query.*;
import org.springframework.stereotype.Repository;
import java.time.Instant;

@Repository
@RequiredArgsConstructor
public class AuthTokenStore {
    private final MongoTemplate mongo;

    // Compare-and-set predicates also protect multiple application instances.
    public boolean consumeVerification(String hash, Instant now) {
        var criteria = Criteria.where("tokenHash").is(hash).and("isDeleted").is(false)
                .and("usedAt").is(null).and("expiresAt").gt(now);
        return mongo.updateFirst(Query.query(criteria),
                new Update().set("usedAt", now).set("updatedAt", now), VerificationToken.class)
                .getModifiedCount() == 1;
    }

    public boolean revokeRefresh(String hash, Instant now) {
        var criteria = Criteria.where("tokenHash").is(hash).and("isDeleted").is(false)
                .and("revokedAt").is(null).and("expiresAt").gt(now);
        return mongo.updateFirst(Query.query(criteria),
                new Update().set("revokedAt", now).set("updatedAt", now), RefreshToken.class)
                .getModifiedCount() == 1;
    }

    public boolean claimResend(String userId, Instant cutoff, Instant now) {
        var criteria = Criteria.where("_id").is(new ObjectId(userId)).and("isDeleted").is(false)
                .and("enabled").is(false).and("emailVerified").is(false)
                .orOperator(Criteria.where("lastVerificationSentAt").lte(cutoff),
                        Criteria.where("lastVerificationSentAt").is(null));
        return mongo.updateFirst(Query.query(criteria),
                new Update().set("lastVerificationSentAt", now).set("updatedAt", now), User.class)
                .getModifiedCount() == 1;
    }

    public void invalidateVerifications(String userId, Instant now) {
        var criteria = Criteria.where("userId").is(new ObjectId(userId))
                .and("isDeleted").is(false).and("usedAt").is(null);
        mongo.updateMulti(Query.query(criteria),
                new Update().set("usedAt", now).set("updatedAt", now), VerificationToken.class);
    }
}
~~~

~~~java
import com.kitolam.rentalroommanagement.common.exception.exceptions.*;
import com.kitolam.rentalroommanagement.modules.auth.config.AuthProperties;
import com.kitolam.rentalroommanagement.modules.auth.document.*;
import com.kitolam.rentalroommanagement.modules.auth.dto.request.*;

import com.kitolam.rentalroommanagement.modules.auth.repository.*;
import com.kitolam.rentalroommanagement.modules.auth.service.*;
import com.kitolam.rentalroommanagement.security.*;
import org.springframework.context.ApplicationEventPublisher;
import org.springframework.security.crypto.password.PasswordEncoder;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;
import java.nio.charset.StandardCharsets;
import java.time.Clock;
import java.time.Duration;

@Service
public class AuthServiceImpl implements AuthService {
    private static final String INVALID_CREDENTIALS = "Email hoặc mật khẩu không đúng";
    private static final String RESEND_MESSAGE =
            "Nếu email có tài khoản chưa xác thực, liên kết xác thực sẽ được gửi. Vui lòng kiểm tra email";
    private final UserRepository users;
    private final VerificationTokenRepository verifications;
    private final RefreshTokenRepository refreshTokens;
    private final AuthTokenStore tokenStore;
    private final PasswordEncoder passwords;
    private final OpaqueTokenProvider tokens;
    private final JwtProvider jwt;
    private final SecurityProperties security;
    private final AuthProperties properties;
    private final ApplicationEventPublisher events;
    private final Clock clock;
    private final String dummyPasswordHash;

    public AuthServiceImpl(UserRepository users, VerificationTokenRepository verifications,
            RefreshTokenRepository refreshTokens, AuthTokenStore tokenStore, PasswordEncoder passwords,
            OpaqueTokenProvider tokens, JwtProvider jwt, SecurityProperties security,
            AuthProperties properties, ApplicationEventPublisher events, Clock clock) {
        this.users = users;
        this.verifications = verifications;
        this.refreshTokens = refreshTokens;
        this.tokenStore = tokenStore;
        this.passwords = passwords;
        this.tokens = tokens;
        this.jwt = jwt;
        this.security = security;
        this.properties = properties;
        this.events = events;
        this.clock = clock;
        // Unknown emails still perform BCrypt work, reducing account-enumeration timing differences.
        this.dummyPasswordHash = passwords.encode(tokens.generate());
    }

    @Override
    @Transactional
    public MessageResponse signup(SignupRequest request) {
        if (users.existsByEmail(request.email())) throw new ConflictException("Email đã được sử dụng");
        if (users.existsByPhone(request.phone())) throw new ConflictException("Số điện thoại đã được sử dụng");
        var now = clock.instant();
        var user = new User();
        user.setFullName(request.fullName());
        user.setEmail(request.email());
        user.setPhone(request.phone());
        user.setPasswordHash(passwords.encode(request.password()));
        user.setRole(Role.TENANT);
        user.setEnabled(false);
        user.setEmailVerified(false);
        user.setTermsAcceptedAt(now);
        user.setLastVerificationSentAt(now);
        users.insert(user);
        sendVerification(user);
        return new MessageResponse("Đăng ký thành công. Vui lòng kiểm tra email để kích hoạt tài khoản");
    }

    @Override
    @Transactional
    public MessageResponse verifyEmail(String token) {
        var stored = verifications.findByTokenHashAndIsDeletedFalse(tokens.hash(token))
                .orElseThrow(() -> new BadRequestException("Token xác thực không hợp lệ"));
        if (!tokens.matches(token, stored.getTokenHash())) throw new BadRequestException("Token xác thực không hợp lệ");
        if (stored.getUsedAt() != null) throw new BadRequestException("Token xác thực đã được sử dụng hoặc thay thế");
        var now = clock.instant();
        if (!stored.getExpiresAt().isAfter(now)) throw new BadRequestException("Token xác thực đã hết hạn");
        if (!tokenStore.consumeVerification(stored.getTokenHash(), now)) {
            throw new BadRequestException("Token xác thực không còn hiệu lực");
        }
        var user = users.findByIdAndIsDeletedFalse(stored.getUserId())
                .orElseThrow(() -> new BadRequestException("Tài khoản không còn tồn tại"));
        user.setEmailVerified(true);
        user.setEnabled(true);
        users.save(user);
        tokenStore.invalidateVerifications(user.getId(), now);
        return new MessageResponse("Xác thực email thành công. Bạn có thể đăng nhập");
    }

    @Override
    @Transactional
    public MessageResponse resendVerification(ResendVerificationRequest request) {
        var user = users.findByEmailAndIsDeletedFalse(request.email()).orElse(null);
        if (user == null || user.isEmailVerified()) return new MessageResponse(RESEND_MESSAGE);
        var now = clock.instant();
        if (!tokenStore.claimResend(user.getId(), now.minus(properties.resendCooldown()), now)) {
            throw new TooManyRequestsException("Vui lòng chờ " + properties.resendCooldown().toSeconds()
                    + " giây giữa hai lần gửi email xác thực");
        }
        tokenStore.invalidateVerifications(user.getId(), now);
        sendVerification(user);
        return new MessageResponse(RESEND_MESSAGE);
    }

    @Override
    @Transactional
    public TokenResponse login(LoginRequest request) {
        var user = users.findByEmailAndIsDeletedFalse(request.email()).orElse(null);
        String supplied = request.password();
        // BCrypt has a 72-byte input limit, including multibyte Unicode passwords.
        boolean tooLong = supplied.getBytes(StandardCharsets.UTF_8).length > 72;
        boolean matches = passwords.matches(tooLong ? "" : supplied,
                user == null ? dummyPasswordHash : user.getPasswordHash());
        if (user == null || tooLong || !matches) throw new UnauthorizedException(INVALID_CREDENTIALS);
        // Only disclose verification status after the password has been checked.
        if (!user.isEmailVerified()) {
            throw new ForbiddenException("Email chưa được xác thực. Hãy kiểm tra email hoặc gửi lại qua /auth/resend-verification");
        }
        if (!user.isEnabled()) throw new ForbiddenException("Tài khoản đã bị vô hiệu hóa");
        return issueTokens(user);
    }

    @Override
    @Transactional
    public TokenResponse refresh(RefreshTokenRequest request) {
        var stored = refreshTokens.findByTokenHashAndIsDeletedFalse(tokens.hash(request.refreshToken()))
                .orElseThrow(() -> new UnauthorizedException("Refresh token không hợp lệ"));
        var now = clock.instant();
        if (!tokens.matches(request.refreshToken(), stored.getTokenHash()) || stored.getRevokedAt() != null
                || !stored.getExpiresAt().isAfter(now)) {
            throw new UnauthorizedException("Refresh token đã hết hạn hoặc bị thu hồi");
        }
        var user = users.findByIdAndIsDeletedFalse(stored.getUserId())
                .orElseThrow(() -> new UnauthorizedException("Refresh token không hợp lệ"));
        if (!user.isEnabled() || !user.isEmailVerified()) throw new UnauthorizedException("Tài khoản không hoạt động");
        // Revocation and insertion commit together. Concurrent requests cannot both rotate the same token.
        if (!tokenStore.revokeRefresh(stored.getTokenHash(), now)) {
            throw new UnauthorizedException("Refresh token đã được sử dụng");
        }
        return issueTokens(user);
    }

    @Override
    @Transactional
    public MessageResponse logout(RefreshTokenRequest request) {
        tokenStore.revokeRefresh(tokens.hash(request.refreshToken()), clock.instant());
        return new MessageResponse("Đăng xuất thành công");
    }

    private void sendVerification(User user) {
        String raw = tokens.generate();
        var token = new VerificationToken();
        token.setUserId(user.getId());
        token.setTokenHash(tokens.hash(raw));
        token.setExpiresAt(clock.instant().plus(properties.verificationTokenTtl()));
        token.setPurgeAt(token.getExpiresAt().plus(Duration.ofDays(7)));
        verifications.insert(token);
        events.publishEvent(new VerificationMailEvent(user.getEmail(), raw));
    }

    private TokenResponse issueTokens(User user) {
        String raw = tokens.generate();
        var token = new RefreshToken();
        token.setUserId(user.getId());
        token.setTokenHash(tokens.hash(raw));
        token.setExpiresAt(clock.instant().plus(security.refreshTokenTtl()));
        token.setPurgeAt(token.getExpiresAt().plus(Duration.ofDays(7)));
        refreshTokens.insert(token);
        return new TokenResponse(jwt.createAccessToken(user), raw, "Bearer",
                security.accessTokenTtl().toSeconds(), security.refreshTokenTtl().toSeconds());
    }
}
~~~

~~~java
package com.kitolam.rentalroommanagement.security;

import jakarta.servlet.*;
import jakarta.servlet.http.*;
import org.springframework.http.HttpHeaders;
import org.springframework.security.authentication.BadCredentialsException;
import org.springframework.security.authentication.UsernamePasswordAuthenticationToken;
import org.springframework.security.core.AuthenticationException;
import org.springframework.security.core.context.SecurityContextHolder;
import org.springframework.security.oauth2.jwt.JwtException;
import org.springframework.web.filter.OncePerRequestFilter;
import java.io.IOException;

public class JwtAuthenticationFilter extends OncePerRequestFilter {
    private final JwtProvider jwt;
    private final AuthUserDetailsService users;
    private final SecurityErrorHandler errors;
    private final boolean requireHttps;

    public JwtAuthenticationFilter(JwtProvider jwt, AuthUserDetailsService users,
                                   SecurityErrorHandler errors, boolean requireHttps) {
        this.jwt = jwt;
        this.users = users;
        this.errors = errors;
        this.requireHttps = requireHttps;
    }

    @Override
    protected void doFilterInternal(HttpServletRequest request, HttpServletResponse response,
                                    FilterChain chain) throws ServletException, IOException {
        if (requireHttps && !request.isSecure()) {
            errors.write(response, 400, "Vui lòng sử dụng HTTPS");
            return;
        }
        // Token responses and verification URLs must not be cached or leaked via referrers.
        response.setHeader("Cache-Control", "no-store");
        response.setHeader("Referrer-Policy", "no-referrer");
        String authorization = request.getHeader(HttpHeaders.AUTHORIZATION);
        if (authorization != null) {
            try {
                if (!authorization.regionMatches(true, 0, "Bearer ", 0, 7)
                        || authorization.length() > 8192) throw new BadCredentialsException("Invalid bearer token");
                var claims = jwt.decode(authorization.substring(7));
                var principal = users.loadById(claims.getSubject());
                if (!principal.isEnabled() || !principal.email().equals(claims.getClaimAsString("email"))
                        || !principal.role().equals(claims.getClaimAsString("role"))) {
                    throw new BadCredentialsException("Inactive account or changed claims");
                }
                var authentication = UsernamePasswordAuthenticationToken.authenticated(
                        principal, null, principal.getAuthorities());
                var context = SecurityContextHolder.createEmptyContext();
                context.setAuthentication(authentication);
                SecurityContextHolder.setContext(context);
            } catch (JwtException | AuthenticationException | IllegalArgumentException ex) {
                SecurityContextHolder.clearContext();
                errors.commence(request, response, new BadCredentialsException("Invalid bearer token"));
                return;
            }
        }
        chain.doFilter(request, response);
    }
}
~~~

~~~java
package com.kitolam.rentalroommanagement.security;

import com.kitolam.rentalroommanagement.modules.auth.document.User;
import com.nimbusds.jose.jwk.source.ImmutableSecret;
import org.springframework.security.oauth2.core.*;
import org.springframework.security.oauth2.jose.jws.MacAlgorithm;
import org.springframework.security.oauth2.jwt.*;
import org.springframework.stereotype.Component;
import javax.crypto.spec.SecretKeySpec;
import java.time.Clock;
import java.time.Duration;
import java.util.Base64;
import java.util.List;
import java.util.UUID;

@Component
public class JwtProvider {
    private final SecurityProperties properties;
    private final Clock clock;
    private final JwtEncoder encoder;
    private final NimbusJwtDecoder decoder;

    public JwtProvider(SecurityProperties properties, Clock clock) {
        this.properties = properties;
        this.clock = clock;
        byte[] key;
        try {
            key = Base64.getDecoder().decode(properties.jwtSecret());
        } catch (IllegalArgumentException ex) {
            throw new IllegalStateException("JWT_SECRET phải là Base64 hợp lệ");
        }
        if (key.length < 32) throw new IllegalStateException("JWT_SECRET cần ít nhất 32 byte ngẫu nhiên");
        var secretKey = new SecretKeySpec(key, "HmacSHA256");
        encoder = new NimbusJwtEncoder(new ImmutableSecret<>(secretKey));
        decoder = NimbusJwtDecoder.withSecretKey(secretKey).macAlgorithm(MacAlgorithm.HS256).build();
        var timestamps = new JwtTimestampValidator(Duration.ZERO);
        timestamps.setClock(clock);
        decoder.setJwtValidator(new DelegatingOAuth2TokenValidator<>(
                timestamps, new JwtIssuerValidator(properties.issuer()), jwt -> {
                    boolean valid = jwt.getExpiresAt() != null && jwt.getIssuedAt() != null
                            && !jwt.getIssuedAt().isAfter(clock.instant())
                            && jwt.getAudience() != null && jwt.getAudience().contains(properties.audience())
                            && jwt.getSubject() != null && jwt.getSubject().matches("[0-9a-f]{24}")
                            && jwt.getSubject().equals(jwt.getClaimAsString("userId"))
                            && jwt.getClaimAsString("email") != null
                            && jwt.getClaimAsString("role") != null
                            && List.of("TENANT", "MANAGER", "ADMIN").contains(jwt.getClaimAsString("role"));
                    return valid ? OAuth2TokenValidatorResult.success() :
                            OAuth2TokenValidatorResult.failure(new OAuth2Error("invalid_token"));
                }));
    }

    public String createAccessToken(User user) {
        var now = clock.instant();
        var claims = JwtClaimsSet.builder().issuer(properties.issuer())
                .audience(List.of(properties.audience())).subject(user.getId())
                .id(UUID.randomUUID().toString()).issuedAt(now)
                .expiresAt(now.plus(properties.accessTokenTtl()))
                .claim("userId", user.getId()).claim("email", user.getEmail())
                .claim("role", user.getRole().name()).build();
        return encoder.encode(JwtEncoderParameters.from(
                JwsHeader.with(MacAlgorithm.HS256).type("JWT").build(), claims)).getTokenValue();
    }

    public Jwt decode(String token) { return decoder.decode(token); }
}
~~~

~~~java
package com.kitolam.rentalroommanagement.security;

import org.springframework.stereotype.Component;
import java.nio.charset.StandardCharsets;
import java.security.MessageDigest;
import java.security.NoSuchAlgorithmException;
import java.security.SecureRandom;
import java.util.Base64;
import java.util.HexFormat;

@Component
public class OpaqueTokenProvider {
    private final SecureRandom random = new SecureRandom();

    public String generate() {
        byte[] bytes = new byte[32];
        random.nextBytes(bytes);
        return Base64.getUrlEncoder().withoutPadding().encodeToString(bytes);
    }

    public String hash(String token) {
        try {
            return HexFormat.of().formatHex(MessageDigest.getInstance("SHA-256")
                    .digest(token.getBytes(StandardCharsets.UTF_8)));
        } catch (NoSuchAlgorithmException ex) {
            throw new IllegalStateException("SHA-256 unavailable");
        }
    }

    public boolean matches(String token, String expectedHash) {
        return MessageDigest.isEqual(hash(token).getBytes(StandardCharsets.US_ASCII),
                expectedHash.getBytes(StandardCharsets.US_ASCII));
    }
}
~~~

~~~java
package com.kitolam.rentalroommanagement.security.validation;

import jakarta.validation.ConstraintValidator;
import jakarta.validation.ConstraintValidatorContext;
import java.nio.charset.StandardCharsets;

public class StrongPasswordValidator implements ConstraintValidator<StrongPassword, String> {
    @Override
    public boolean isValid(String value, ConstraintValidatorContext context) {
        if (value == null || value.codePointCount(0, value.length()) < 8
                || value.getBytes(StandardCharsets.UTF_8).length > 72) return false;
        return value.codePoints().anyMatch(Character::isUpperCase)
                && value.codePoints().anyMatch(Character::isLowerCase)
                && value.codePoints().anyMatch(Character::isDigit)
                && value.codePoints().anyMatch(c -> !Character.isLetterOrDigit(c) && !Character.isWhitespace(c))
                && value.codePoints().noneMatch(Character::isISOControl);
    }
}
~~~
